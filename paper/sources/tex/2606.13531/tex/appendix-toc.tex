\makeatletter
\newif\ifappendixcontentsactive
\appendixcontentsactivefalse

\let\appendixcontents@origaddcontentsline\addcontentsline
\renewcommand{\addcontentsline}[3]{%
  \appendixcontents@origaddcontentsline{#1}{#2}{#3}%
  \ifappendixcontentsactive
  \def\appendixcontents@ext{#1}%
  \def\appendixcontents@toc{toc}%
  \def\appendixcontents@unit{#2}%
  \def\appendixcontents@section{section}%
  \ifx\appendixcontents@ext\appendixcontents@toc
  \ifx\appendixcontents@unit\appendixcontents@section
  \appendixcontents@origaddcontentsline{atoc}{#2}{#3}%
  \fi
  \fi
  \fi
}

\newcommand{\printappendixcontents}{%
  \@ifclassloaded{acmart}{%
    \@starttoc{atoc}{Appendix Contents}%
  }{%
    \section*{Appendix Contents}%
    \@starttoc{atoc}%
  }%
  \appendixcontentsactivetrue
}

\newcommand{\stopappendixcontents}{%
  \appendixcontentsactivefalse
}
\makeatother
